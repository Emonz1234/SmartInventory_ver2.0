"""Keep PySide6 and PyQt6 in different processes, as deployed."""
import importlib.util
from pathlib import Path
import subprocess
import sys
import pytest


@pytest.mark.skipif(importlib.util.find_spec('PySide6') is None, reason='Install desktop-requirements.txt')
def test_pyside_console_data_controls_and_shutdown():
    script = r'''
import os, time
os.environ['QT_QPA_PLATFORM']='offscreen'
from types import SimpleNamespace
from PySide6.QtWidgets import QApplication
from ipc_core.desktop import EdgeWindow
from unittest.mock import patch
cfg=SimpleNamespace(DEVICE_ID='test-sim', DEVICE_TYPE='IPCSIM', EDGE_API_HOST='127.0.0.1', EDGE_API_PORT=1,
                    EDGE_API_TOKEN='test', SERVER_URL='')
app=QApplication([])
with patch('ipc_core.desktop.read_edge', return_value={'health': {}, 'error':'test offline'}):
    window=EdgeWindow(cfg)
    window.show()
    window.display({'health': {'server_synced': True,'serial_connected':True}, 'racks': {'1': {'temperature':22}},
      'records':[{'kind':'rack','data':{'id':4,'rack_code':'1','rack_name':'Rack'}},
                 {'kind':'item','data':{'id':1,'item_name':'Part'}}, {'kind':'bin','data':{'id':2,'bin_code':'B'}}],
      'pending':[], 'events':[]})
    assert not any(b.isEnabled() for b in window.buttons)
    window.auth_changed(True)
    assert all(b.isEnabled() for b in window.buttons)
    window.health = {}
    window.update_controls()
    assert not any(b.isEnabled() for b in window.buttons)
    window.close()
    deadline=time.monotonic()+5
    while window.worker.isRunning() and time.monotonic()<deadline:
        app.processEvents();time.sleep(.01)
    assert not window.worker.isRunning()
    window.close()
'''
    result = subprocess.run([sys.executable, '-c', script], cwd=Path(__file__).resolve().parents[1],
                            capture_output=True, text=True, timeout=15)
    assert result.returncode == 0, result.stdout + result.stderr
